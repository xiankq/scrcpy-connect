cmake_minimum_required(VERSION 3.18)
project(media_codec_core LANGUAGES C CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/ffmpeg.cmake")

find_program(MEDIA_CODEC_CARGO cargo HINTS "$ENV{CARGO_HOME}/bin" "$ENV{HOME}/.cargo/bin" "$ENV{USERPROFILE}/.cargo/bin" REQUIRED)
set(MEDIA_CODEC_RUST_ROOT "${MEDIA_CODEC_WORKSPACE_ROOT}/rust/media_codec")
set(MEDIA_CODEC_RUST_TARGET_DIR "${MEDIA_CODEC_WORKSPACE_ROOT}/.dart_tool/media_codec/rust-target" CACHE PATH "Shared media decoder Cargo cache")
set(MEDIA_CODEC_RUST_ENV "MEDIA_CODEC_FFMPEG_ROOT=${MEDIA_CODEC_FFMPEG_ROOT}" "CARGO_INCREMENTAL=0" "MAKEFLAGS=")

if(ANDROID)
  if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(MEDIA_CODEC_RUST_TARGET aarch64-linux-android)
    set(MEDIA_CODEC_CLANG_TARGET aarch64-linux-android)
  elseif(ANDROID_ABI STREQUAL "armeabi-v7a")
    set(MEDIA_CODEC_RUST_TARGET armv7-linux-androideabi)
    set(MEDIA_CODEC_CLANG_TARGET armv7a-linux-androideabi)
  elseif(ANDROID_ABI STREQUAL "x86_64")
    set(MEDIA_CODEC_RUST_TARGET x86_64-linux-android)
    set(MEDIA_CODEC_CLANG_TARGET x86_64-linux-android)
  else()
    message(FATAL_ERROR "Unsupported Flutter Android ABI: ${ANDROID_ABI}")
  endif()
  get_filename_component(MEDIA_CODEC_NDK_BIN "${CMAKE_C_COMPILER}" DIRECTORY)
  if(NOT ANDROID_PLATFORM_LEVEL)
    string(REGEX REPLACE "android-" "" ANDROID_PLATFORM_LEVEL "${ANDROID_PLATFORM}")
  endif()
  set(MEDIA_CODEC_RUST_LINKER "${MEDIA_CODEC_NDK_BIN}/${MEDIA_CODEC_CLANG_TARGET}${ANDROID_PLATFORM_LEVEL}-clang")
  if(CMAKE_HOST_WIN32)
    string(APPEND MEDIA_CODEC_RUST_LINKER ".cmd")
  endif()
  list(APPEND MEDIA_CODEC_RUST_ENV "MEDIA_CODEC_SYSROOT=${MEDIA_CODEC_NDK_BIN}/../sysroot")
  if(NOT DEFINED ENV{LIBCLANG_PATH})
    if(CMAKE_HOST_WIN32)
      list(APPEND MEDIA_CODEC_RUST_ENV "LIBCLANG_PATH=${MEDIA_CODEC_NDK_BIN}")
    else()
      list(APPEND MEDIA_CODEC_RUST_ENV "LIBCLANG_PATH=${MEDIA_CODEC_NDK_BIN}/../lib")
    endif()
  endif()
elseif(WIN32)
  if(CMAKE_GENERATOR_PLATFORM MATCHES "ARM64" OR CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|ARM64|arm64")
    set(MEDIA_CODEC_RUST_TARGET aarch64-pc-windows-msvc)
  else()
    set(MEDIA_CODEC_RUST_TARGET x86_64-pc-windows-msvc)
  endif()
elseif(APPLE)
  if(CMAKE_OSX_ARCHITECTURES)
    list(LENGTH CMAKE_OSX_ARCHITECTURES MEDIA_CODEC_ARCH_COUNT)
    if(MEDIA_CODEC_ARCH_COUNT GREATER 1)
      message(FATAL_ERROR "Build each Apple architecture separately, then combine using lipo; CocoaPods handles the universal build automatically")
    endif()
    set(MEDIA_CODEC_APPLE_ARCH "${CMAKE_OSX_ARCHITECTURES}")
  else()
    set(MEDIA_CODEC_APPLE_ARCH "${CMAKE_SYSTEM_PROCESSOR}")
  endif()
  if(MEDIA_CODEC_APPLE_ARCH MATCHES "arm64|aarch64")
    set(MEDIA_CODEC_APPLE_RUST_ARCH aarch64)
  else()
    set(MEDIA_CODEC_APPLE_RUST_ARCH x86_64)
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    if(CMAKE_OSX_SYSROOT MATCHES "simulator" AND MEDIA_CODEC_APPLE_RUST_ARCH STREQUAL "aarch64")
      set(MEDIA_CODEC_RUST_TARGET aarch64-apple-ios-sim)
    else()
      set(MEDIA_CODEC_RUST_TARGET "${MEDIA_CODEC_APPLE_RUST_ARCH}-apple-ios")
    endif()
  else()
    set(MEDIA_CODEC_RUST_TARGET "${MEDIA_CODEC_APPLE_RUST_ARCH}-apple-darwin")
  endif()
  if(IS_DIRECTORY "${CMAKE_OSX_SYSROOT}")
    list(APPEND MEDIA_CODEC_RUST_ENV "MEDIA_CODEC_SYSROOT=${CMAKE_OSX_SYSROOT}")
  endif()
else()
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
    set(MEDIA_CODEC_RUST_TARGET aarch64-unknown-linux-gnu)
  else()
    set(MEDIA_CODEC_RUST_TARGET x86_64-unknown-linux-gnu)
  endif()
  if(MC_VA_LIBRARY_DIRS)
    list(JOIN MC_VA_LIBRARY_DIRS ":" MEDIA_CODEC_VA_LIBRARY_PATH)
    list(APPEND MEDIA_CODEC_RUST_ENV "LIBRARY_PATH=${MEDIA_CODEC_VA_LIBRARY_PATH}:$ENV{LIBRARY_PATH}")
  endif()
endif()
if(MEDIA_CODEC_RUST_LINKER)
  string(TOUPPER "${MEDIA_CODEC_RUST_TARGET}" MEDIA_CODEC_RUST_TARGET_KEY)
  string(REPLACE "-" "_" MEDIA_CODEC_RUST_TARGET_KEY "${MEDIA_CODEC_RUST_TARGET_KEY}")
  list(APPEND MEDIA_CODEC_RUST_ENV "CARGO_TARGET_${MEDIA_CODEC_RUST_TARGET_KEY}_LINKER=${MEDIA_CODEC_RUST_LINKER}")
endif()

if(WIN32)
  set(MEDIA_CODEC_RUST_LIBRARY "${MEDIA_CODEC_RUST_TARGET_DIR}/${MEDIA_CODEC_RUST_TARGET}/release/media_codec_native.lib")
else()
  set(MEDIA_CODEC_RUST_LIBRARY "${MEDIA_CODEC_RUST_TARGET_DIR}/${MEDIA_CODEC_RUST_TARGET}/release/libmedia_codec_native.a")
endif()
file(GLOB_RECURSE MEDIA_CODEC_RUST_SOURCES CONFIGURE_DEPENDS "${MEDIA_CODEC_RUST_ROOT}/src/*.rs")
add_custom_command(OUTPUT "${MEDIA_CODEC_RUST_LIBRARY}"
  COMMAND "${CMAKE_COMMAND}" -E env ${MEDIA_CODEC_RUST_ENV} "${MEDIA_CODEC_CARGO}" build
    --manifest-path "${MEDIA_CODEC_RUST_ROOT}/Cargo.toml" --release --locked --lib
    --target "${MEDIA_CODEC_RUST_TARGET}" --target-dir "${MEDIA_CODEC_RUST_TARGET_DIR}"
  DEPENDS ${MEDIA_CODEC_RUST_SOURCES} "${MEDIA_CODEC_RUST_ROOT}/build.rs"
    "${MEDIA_CODEC_RUST_ROOT}/Cargo.toml" "${MEDIA_CODEC_RUST_ROOT}/Cargo.lock"
    "${CMAKE_CURRENT_SOURCE_DIR}/media_codec.h" "${MEDIA_CODEC_FFMPEG_ROOT}/media-codec-build.json"
  COMMENT "Building Rust packet decoder for ${MEDIA_CODEC_RUST_TARGET}"
  WORKING_DIRECTORY "${MEDIA_CODEC_RUST_ROOT}"
  VERBATIM)
add_custom_target(media_codec_rust_build DEPENDS "${MEDIA_CODEC_RUST_LIBRARY}")
add_library(media_codec_core STATIC IMPORTED GLOBAL)
set_target_properties(media_codec_core PROPERTIES IMPORTED_LOCATION "${MEDIA_CODEC_RUST_LIBRARY}"
  INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}")
add_dependencies(media_codec_core media_codec_rust_build)
target_link_libraries(media_codec_core INTERFACE mc_ffmpeg)
find_package(Threads REQUIRED)
target_link_libraries(media_codec_core INTERFACE Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
  target_link_libraries(media_codec_core INTERFACE ws2_32 userenv ntdll advapi32)
endif()
option(MEDIA_CODEC_BUILD_TESTS "Build native lifecycle tests" OFF)
if(MEDIA_CODEC_BUILD_TESTS)
  enable_testing()
  add_executable(media_codec_core_test tests/core_test.cpp)
  target_link_libraries(media_codec_core_test PRIVATE media_codec_core)
  if(MSVC)
    target_compile_options(media_codec_core_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(media_codec_core_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME media_codec_core_test COMMAND media_codec_core_test)
endif()
