cmake_minimum_required(VERSION 3.14)
project(media_codec_core LANGUAGES C CXX)
if(NOT TARGET mc_ffmpeg AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../cmake/ffmpeg.cmake")
  include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/ffmpeg.cmake")
endif()
set(MEDIA_CODEC_FFMPEG_ROOT "$ENV{MEDIA_CODEC_FFMPEG_ROOT}" CACHE PATH "Trimmed FFmpeg prefix")
if(NOT TARGET mc_ffmpeg AND NOT MEDIA_CODEC_FFMPEG_ROOT)
  message(FATAL_ERROR "MEDIA_CODEC_FFMPEG_ROOT must point to the trimmed FFmpeg prefix; run the repository media dependency builder")
endif()
add_library(media_codec_core STATIC media_codec.cpp)
set_target_properties(media_codec_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(media_codec_core PUBLIC cxx_std_17)
target_include_directories(media_codec_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${MEDIA_CODEC_FFMPEG_ROOT}/include")
if(TARGET mc_ffmpeg)
  target_link_libraries(media_codec_core PUBLIC mc_ffmpeg)
else()
  foreach(component avcodec avutil swscale)
    find_library(MEDIA_CODEC_${component}_LIBRARY NAMES ${component} lib${component} PATHS "${MEDIA_CODEC_FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(media_codec_core PUBLIC "${MEDIA_CODEC_${component}_LIBRARY}")
  endforeach()
endif()
find_package(Threads REQUIRED)
target_link_libraries(media_codec_core PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
if(UNIX AND NOT APPLE)
  target_link_libraries(media_codec_core PUBLIC m)
endif()
if(WIN32)
  target_link_libraries(media_codec_core PUBLIC bcrypt ole32 uuid dxva2 d3d11)
endif()
option(MEDIA_CODEC_BUILD_TESTS "Build native lifecycle tests" OFF)
if(MEDIA_CODEC_BUILD_TESTS)
  enable_testing()
  add_executable(media_codec_core_test tests/core_test.cpp)
  target_link_libraries(media_codec_core_test PRIVATE media_codec_core)
  add_test(NAME media_codec_core_test COMMAND media_codec_core_test)
endif()
